<script>
    const distancias = [3.0, 3.0, 3.0, 3.0, 3.0, 3.0, 3.1];

    const estrategias = {
        sub135: ["04:35", "04:28", "04:28", "04:38", "04:28", "04:25", "04:18"],
        sub145: ["05:05", "04:56", "04:56", "05:08", "04:56", "04:52", "04:45"],
        sub200: ["05:50", "05:38", "05:38", "05:52", "05:38", "05:35", "05:25"],
        sub210: ["06:15", "06:05", "06:05", "06:20", "06:05", "06:00", "05:50"],
        sub225: ["07:00", "06:50", "06:50", "07:05", "06:50", "06:45", "06:30"]
    };

    function cargarMeta(clave, btnElement) {
        const ritmos = estrategias[clave];
        if (!ritmos) return;
        for (let i = 1; i <= 7; i++) {
            document.getElementById(`pace${i}`).value = ritmos[i - 1];
        }
        const botones = document.querySelectorAll('.preset-btn');
        botones.forEach(b => b.classList.remove('active'));
        if (btnElement) btnElement.classList.add('active');
        calcularTiempos();
    }

    function timeToSeconds(timeStr) {
        if (!timeStr) return 0;
        const parts = timeStr.trim().split(':');
        if (parts.length !== 2) return 0;
        const m = parseInt(parts[0], 10);
        const s = parseInt(parts[1], 10);
        if (isNaN(m) || isNaN(s)) return 0;
        return (m * 60) + s;
    }

    function secondsToTime(totalSeconds, includeHours = false) {
        const rounded = Math.round(totalSeconds);
        const h = Math.floor(rounded / 3600);
        const m = Math.floor((rounded % 3600) / 60);
        const s = rounded % 60;
        const mStr = m.toString().padStart(2, '0');
        const sStr = s.toString().padStart(2, '0');
        if (includeHours || h > 0) {
            return `${h.toString().padStart(2, '0')}:${mStr}:${sStr}`;
        }
        return `${mStr}:${sStr}`;
    }

    function calcularTiempos() {
        let totalRaceSeconds = 0;
        for (let i = 1; i <= 7; i++) {
            const paceInput = document.getElementById(`pace${i}`).value;
            const paceSeconds = timeToSeconds(paceInput);
            const segmentTimeSeconds = paceSeconds * distancias[i - 1];
            totalRaceSeconds += segmentTimeSeconds;
            document.getElementById(`time${i}`).textContent = secondsToTime(segmentTimeSeconds);
            document.getElementById(`acc${i}`).textContent = secondsToTime(totalRaceSeconds, true);
        }
        document.getElementById('totalTime').textContent = secondsToTime(totalRaceSeconds, true);
        const avgPaceSeconds = totalRaceSeconds / 21.1;
        document.getElementById('avgPace').textContent = secondsToTime(avgPaceSeconds) + " /km";
    }
    
    window.onload = calcularTiempos;
</script>
